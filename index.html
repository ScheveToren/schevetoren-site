// Google Apps Script template for Scheve Toren attendance backend. // Deploy as a web app and set the public URL in the attendance page config. // This handles Lichess OAuth callback, storing invite codes, attendance rows and CSV export. var SPREADSHEET_ID = "PUT_YOUR_GOOGLE_SHEET_ID_HERE"; var PLAYERS_SHEET = "Players"; var CODES_SHEET = "InviteCodes"; var ATTENDANCE_SHEET = "Attendance"; function ensureSheets() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheets = ss.getSheets().map(function (sheet) { return sheet.getName(); }); if (!sheets.includes(PLAYERS_SHEET)) { ss.insertSheet(PLAYERS_SHEET); var playersSheet = ss.getSheetByName(PLAYERS_SHEET); playersSheet.appendRow(["Player ID", "Name", "Lichess Username", "Invite Code", "Linked", "Active"]); } if (!sheets.includes(CODES_SHEET)) { ss.insertSheet(CODES_SHEET); var codesSheet = ss.getSheetByName(CODES_SHEET); codesSheet.appendRow(["Code", "Player ID", "Used", "Created At", "Used At"]); } if (!sheets.includes(ATTENDANCE_SHEET)) { ss.insertSheet(ATTENDANCE_SHEET); var attendanceSheet = ss.getSheetByName(ATTENDANCE_SHEET); attendanceSheet.appendRow(["Season", "Date", "Player ID", "Player Name", "Status", "Note", "Updated At"]); } } function doGet(e) { ensureSheets(); var action = (e && e.parameter && e.parameter.action) || "status"; if (action === "players") { return ContentService.createTextOutput(JSON.stringify(getPlayers())); } if (action === "attendance") { return ContentService.createTextOutput(JSON.stringify(getAttendanceForSeason(e.parameter.season || "2026-2027"))); } if (action === "export") { return ContentService.createTextOutput(exportAttendanceCsv(e.parameter.season || "2026-2027")); } return ContentService.createTextOutput(JSON.stringify({ ok: true, message: "Scheve Toren attendance backend ready." })); } function doPost(e) { ensureSheets(); var payload = JSON.parse(e.postData.contents || "{}"); var action = payload.action || "saveAttendance"; if (action === "linkPlayer") { return ContentService.createTextOutput(JSON.stringify(linkPlayerByInviteCode(payload))); } if (action === "saveAttendance") { return ContentService.createTextOutput(JSON.stringify(saveAttendance(payload))); } return ContentService.createTextOutput(JSON.stringify({ ok: false, message: "Unknown action" })); } function getPlayers() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(PLAYERS_SHEET); var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var heading = values[0]; var rows = []; for (var i = 1; i < values.length; i++) { var row = values[i]; if (!row[0]) continue; var player = {}; for (var j = 0; j < heading.length; j++) { player[heading[j]] = row[j] || ""; } rows.push(player); } return rows; } function getAttendanceForSeason(season) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ATTENDANCE_SHEET); var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var rows = []; for (var i = 1; i < values.length; i++) { var row = values[i]; if (row[0] === season) { rows.push({ season: row[0], date: row[1], playerId: row[2], playerName: row[3], status: row[4], note: row[5], updatedAt: row[6] }); } } return rows; } function linkPlayerByInviteCode(payload) { var code = String(payload.code || "").trim(); var lichessUser = String(payload.lichessUser || "").trim(); if (!code || !lichessUser) { return { ok: false, message: "Code and Lichess username required." }; } var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CODES_SHEET); var values = sheet.getDataRange().getValues(); var found = false; for (var i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === code) { found = true; if (String(values[i][2]).trim().toLowerCase() === "true") { return { ok: false, message: "Code is already used." }; } var playerId = values[i][1]; var playersSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(PLAYERS_SHEET); var playerRows = playersSheet.getDataRange().getValues(); for (var j = 1; j < playerRows.length; j++) { if (String(playerRows[j][0]).trim() === String(playerId)) { playersSheet.getRange(j + 1, 3).setValue(lichessUser); playersSheet.getRange(j + 1, 4).setValue(code); playersSheet.getRange(j + 1, 5).setValue("TRUE"); break; } } sheet.getRange(i + 1, 3).setValue("TRUE"); sheet.getRange(i + 1, 5).setValue(new Date().toISOString()); return { ok: true, message: "Player linked successfully.", playerId: playerId }; } } if (!found) { return { ok: false, message: "Invite code not found." }; } return { ok: false, message: "Failed to link player." }; } function saveAttendance(payload) { var season = payload.season || "2026-2027"; var playerName = payload.playerName || ""; var date = payload.date || ""; var status = payload.status || "absent"; var note = payload.note || ""; var updatedAt = new Date().toISOString(); if (!season || !playerName || !date) { return { ok: false, message: "Season, player name and date are required." }; } var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ATTENDANCE_SHEET); var rows = sheet.getDataRange().getValues(); var found = false; for (var i = 1; i < rows.length; i++) { if (rows[i][0] === season && rows[i][1] === date && rows[i][3] === playerName) { sheet.getRange(i + 1, 5).setValue(status); sheet.getRange(i + 1, 6).setValue(note); sheet.getRange(i + 1, 7).setValue(updatedAt); found = true; break; } } if (!found) { sheet.appendRow([season, date, "", playerName, status, note, updatedAt]); } return { ok: true, message: "Attendance saved." }; } function exportAttendanceCsv(season) { var rows = getAttendanceForSeason(season); var content = ["season,date,playerName,status,note"]; rows.forEach(function (row) { content.push([ row.season, row.date, row.playerName, row.status, (row.note || "").replace(/\"/g, '\"\"') ].join(",")); }); return content.join("\n"); } // Example of a generated invite code: // function createInviteCode(playerId) { // var code = "ST-" + Utilities.getUuid().slice(0, 8).toUpperCase(); // var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CODES_SHEET);
